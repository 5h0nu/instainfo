Fetch Profile Info

Send a GET request to query profile details programmatically.

GET /api/instagram/profile?username={username}
fetch(`/api/instagram/profile?username=nasa`) .then(res => res.json()) .then(data => { if (data.success) { console.log(`Followers: ${data.followers_count}`); console.log(`Bio: ${data.bio}`); } else { console.error("Error:", data.error); } });
import requests url = "https://your-domain.com/api/instagram/profile" params = {"username": "nasa"} response = requests.get(url, params=params) data = response.json() if data.get("success"): print(f"Followers: {data['followers_count']}") print(f"Bio: {data['bio']}") else: print(f"Error: {data.get('error')}")
